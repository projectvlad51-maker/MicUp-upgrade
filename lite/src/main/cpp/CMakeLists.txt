cmake_minimum_required(VERSION 3.22)
project(micup_lite_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(oboe REQUIRED CONFIG)
add_library(micplugin_native SHARED ../../../../app/src/main/cpp/AudioEngine.cpp)
target_compile_definitions(micplugin_native PRIVATE MICUP_LITE_NATIVE ANDROID)
target_compile_options(micplugin_native PRIVATE -O2 -ffast-math -Wall -Wno-unused-parameter)
target_link_libraries(micplugin_native oboe::oboe log android OpenSLES)
